## 一、项目概述

### 1.1 项目目标

构建一套面向垂直领域技能培训的**领域知识个性化生成与多智能体协同决策系统**。系统通过多个 AI Agent 协同工作，实现对不同背景学习者的精准诊断，并动态生成高保真、个性化的学习资源（定制讲义、实操指南、分阶测试题等），解决大模型在垂直领域中"幻觉不可控、个性化与专业性难以兼顾"的核心痛点。

### 1.2 核心指标

| 指标 | 目标值 |
|------|--------|
| 专业知识谬误率（幻觉率） | < 5% |
| 学习者画像-资源难度匹配准确率 | ≥ 85% |
| 知识点覆盖率 | ≥ 90% |

### 1.3 评分权重

| 评分维度 | 分值 | 关键要求 |
|----------|------|----------|
| 作品完整性 | 30 分 | "学情画像→协同调度→知识生成→交互反馈→动态决策"全流程闭环 |
| 技术创新性 | 25 分 | 多 Agent 交叉验证消除幻觉、突破传统静态资源分发模式 |
| 用户体验 | 15 分 | 界面简洁、流程便捷、协同过程可视化、资源排版规范 |
| 实用价值 | 30 分 | 知识准确贴合行业规范、指标达标、方案可泛化迁移 |

```python
# backend/app/knowledge/loader.py 示例
from langchain_community.document_loaders import DirectoryLoader, TextLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import HuggingFaceEmbeddings

class KnowledgeBase:
    def __init__(self, persist_dir: str = "./chroma_db"):
        self.embeddings = HuggingFaceEmbeddings(
            model_name="BAAI/bge-m3",
            model_kwargs={'device': 'cpu'}
        )
        self.vectorstore = None
        self.persist_dir = persist_dir
    
    def load_documents(self, doc_dir: str):
        loader = DirectoryLoader(
            doc_dir, 
            glob="**/*.md",
            loader_cls=TextLoader
        )
        documents = loader.load()
        
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000,
            chunk_overlap=200
        )
        chunks = splitter.split_documents(documents)
        
        self.vectorstore = Chroma.from_documents(
            documents=chunks,
            embedding=self.embeddings,
            persist_directory=self.persist_dir
        )
        return len(chunks)
    
    def retrieve(self, query: str, k: int = 5):
        return self.vectorstore.similarity_search(query, k=k)
```

```python
# backend/app/graph/workflow.py
from langgraph.graph import StateGraph, END
from typing import TypedDict, List, Annotated
import operator

class AgentState(TypedDict):
    # 输入
    learner_input: dict
    topic: str
    # 中间状态
    profile: LearnerProfile
    generated_content: dict  # {type: content}
    review_results: dict  # {type: ReviewResult}
    retry_count: int
    # 输出
    final_resources: List[dict]
    visualization_data: dict
    # 反馈循环
    feedback_history: List[dict]
    decision_log: List[str]

def build_workflow() -> StateGraph:
    workflow = StateGraph(AgentState)
    
    # 添加节点
    workflow.add_node("diagnose", diagnose_node)
    workflow.add_node("generate_lecture", generate_lecture_node)
    workflow.add_node("generate_guide", generate_guide_node)
    workflow.add_node("generate_test", generate_test_node)
    workflow.add_node("review_content", review_content_node)
    workflow.add_node("decide_next", decide_next_node)
    workflow.add_node("finalize", finalize_node)
    
    # 定义边（流程）
    workflow.set_entry_point("diagnose")
    workflow.add_edge("diagnose", "generate_lecture")
    workflow.add_edge("diagnose", "generate_guide")
    workflow.add_edge("diagnose", "generate_test")
    # 所有生成节点汇聚到审核
    workflow.add_edge("generate_lecture", "review_content")
    workflow.add_edge("generate_guide", "review_content")
    workflow.add_edge("generate_test", "review_content")
    # 审核 → 决策（重新生成或完成）
    workflow.add_conditional_edges(
        "review_content",
        decide_next_node,
        {
            "retry": "generate_lecture",  # 打回重生成
            "complete": "finalize",
        }
    )
    workflow.add_edge("finalize", END)
    
    return workflow.compile()
```

**条件路由逻辑：**

```python
def decide_next_node(state: AgentState) -> str:
    """决策调度：审核通过则完成，否则打回重新生成"""
    all_passed = all(
        r.passed for r in state["review_results"].values()
    )
    
    if all_passed:
        state["decision_log"].append("所有资源审核通过")
        return "complete"
    
    if state["retry_count"] >= 3:
        state["decision_log"].append("达到最大重试次数，强制完成")
        return "complete"
    
    state["retry_count"] += 1
    state["decision_log"].append(
        f"审核未通过(第{state['retry_count']}次)，打回重新生成。问题: {state['review_results']}"
    )
    return "retry"
```

#### 4.3.1 交互反馈处理

```python
# backend/app/agents/orchestrator.py
class DecisionOrchestrator:
    def handle_feedback(self, state: AgentState, 
                        user_answer: dict) -> AgentState:
        """处理学习者答题反馈，动态调整策略"""
        
        correctness = user_answer["correctness"]  # 正确率
        topic = user_answer["topic"]
        
        if correctness < 0.6:
            # 触发"降维解释"——生成更基础的内容
            state["decision_log"].append(
                f"正确率{correctness:.0%}<60%，触发降维解释"
            )
            state["topic"] = f"{topic} 基础概念"
            state["profile"].knowledge_points = self._downgrade_level(
                state["profile"].knowledge_points, topic
            )
            
        elif correctness > 0.9:
            # 触发"进阶挑战"——生成更高难度的内容
            state["decision_log"].append(
                f"正确率{correctness:.0%}>90%，触发进阶挑战"
            )
            state["topic"] = f"{topic} 高级应用"
```

```typescript
// frontend/composables/useAgentWebSocket.ts
export function useAgentWebSocket(sessionId: string) {
  const config = useRuntimeConfig()
  const agents = ref<AgentStatus[]>([])
  const isConnected = ref(false)

  const ws = new WebSocket(`${config.public.wsBase}/ws/agent-status/${sessionId}`)

  ws.onopen = () => {
    isConnected.value = true
  }

  ws.onmessage = (event) => {
    const data = JSON.parse(event.data)
    const idx = agents.value.findIndex(a => a.name === data.agent)
    if (idx >= 0) {
      agents.value[idx] = { ...agents.value[idx], ...data }
    } else {
      agents.value.push(data)
    }
  }

  ws.onclose = () => {
    isConnected.value = false
  }

  onUnmounted(() => {
    ws.close()
  })

  return { agents, isConnected }
}
```

```python
def cross_validation_debate(
    content: str, 
    topic: str, 
    llm_a,  # 扮演"学术审查者"
    llm_b,  # 扮演"工业实践者"
    knowledge_base

    """
    双角色辩论式验证：
    - 学术审查者：检查理论准确性、概念定义、引用正确性
    - 工业实践者：检查实操可行性、行业规范、代码可运行性
    取两者问题并集，大幅降低漏检率
    """
    perspective_a = llm_a.invoke(f"""

    {content}
    """)

    你是一位有10年行业经验的实践专家。请审查以下内容的实操可行性：
    1. 操作步骤是否可以在真实环境中复现

    3. 是否符合行业规范
```
